2: // GVS5H-5 — deterministic 5-role batch protocol
3: // ============================================================================
4: // One batch = 5 ROLES (the user may read "5 subagents" either way; roles are
5: // the stable unit, invocations repeat inside the repair loop):
6: //
7: //   G1 G2 G3  gatherers  — read spec + local code + context7 + websearch
8: //                          (run in parallel)
9: //   B1 B2 B3  debaters   — argue the approach from the gathered evidence; run
10: //                          CONCURRENTLY with the doer's draft pass
11: //   D          doer       — drafts the approach, then implements it
12: //   R1 R2 R3  reviewers  — audit the finished work
13: //   E          e2e        — end-to-end verification as code AND as a user
14: //
15: // Five role families (gather / debate / do / review / e2e). G, B and R each run
16: // 3 seats, so one batch is 3+3+1+3+1 = 11 agent invocations per iteration.
17: //
18: // Timeline of one batch (all stages are per-item, no cross-batch barrier):
19: //
20: //   gather(x3, parallel)
21: //        |
22: //        +--> doer DRAFT  .............. running in parallel with ...
23: //        |                              debate(x3, parallel)
24: //        +--> doer FINAL (sees gathered findings + debate conclusions)
25: //                  |
26: //                  +--> review(x3) + e2e(1)   [parallel]
27: //                          |
28: //                    any failure? --> repair D --> review again  (loop)
29: //                          |
30: //                    all clean --> batch done
31: //
32: // args (JSON, passed by the Manager):
33: //   {
34: //     spec: string,            // the spec-kit text every agent must obey
35: //     batches: [{ id, title, focus, paths?: string[] }],
36: //     maxIterations?: number,  // repair-loop ceiling, default 3
37: //     debateRounds?: number    // reserved, default 1
38: //   }

40: // Returns: { batches: [...], stats: {...} }

43: const A = args || {};

45: // Normalize args.batches. Some transports deliver a JSON array as a
46: // single-key object ({item: {...}} / {"0": {...}}) instead of a real array, so
47: // accept every shape that still identifies exactly one batch per entry.
48: function normalizeBatches(raw) {
49:   if (raw === undefined || raw === null) return [];
50:   if (Array.isArray(raw)) return raw.filter((x) => x && typeof x === "object");
51:   if (typeof raw === "object") {
52:     // An object carrying a batch `id`/`title` is one batch.
53:     if (raw.id !== undefined || raw.title !== undefined) return [raw];
54:     const keys = Object.keys(raw);
55:     // Keys that look like an unwrapped array: "0","1",... or a lone "item".
56:     const isIndexish = keys.every((k) => k === "item" || /^\d+$/.test(k));
57:     if (isIndexish && keys.length > 0) {
58:       return keys
59:         .sort((a, b) => (a === "item" ? -1 : b === "item" ? 1 : Number(a) - Number(b)))
60:         .map((k) => raw[k])
61:         .filter((x) => x && typeof x === "object");
62:     }
63:   }
64:   return [];
65: }
66: // A batch is only usable if it names itself and states what to do. Anything
67: // else (a transport-mangled wrapper, an empty entry) is dropped loudly rather
68: // than becoming a silent no-op batch that fails deep inside the pipeline.
69: function usable(b) {
70:   return !!(b && typeof b === "object" && (b.id !== undefined || b.title !== undefined) &&
71:             (typeof b.focus === "string" ? b.focus.trim().length > 0 : false));
72: }

224: // ── Stage: review (3 reviewers + 1 e2e, parallel) ───────────────────────────
225: function reviewStage(b, work, seat) {
226:   const base =
227:     specBlock() + batchBlock(b) +
228:     "THE DOER'S REPORT:\nsummary: " + (work ? work.summary : "(none)") +

233:     return agent(
234:       base +
235:         "You are REVIEWER " + (seat + 1) + " of 3. Audit the actual work in the repo — open the changed files, " +
236:         "run the tests yourself, and hunt for: wrong logic, missed spec requirements, broken edge cases, " +
237:         "unhandled errors, security issues, and claims the doer did not actually verify. " +
238:         "Do NOT trust the report — check it. verdict=fail if you find ANY real defect.\n\n" + toolchain(),
239:       { schema: REVIEW, label: "b" + b.id + ":review" + (seat + 1), phase: "review" }
240:     );
241:   }
242:   return agent(
243:     base +
244:       "You are the END-TO-END reviewer. Everything must work BOTH as code AND as a user:\n" +

246:       "- As a user: exercise the real flow end-to-end. If it is a web app, drive it with camofox " +

255: function repairStage(b, work, reviews) {
256:   const fails = reviews.filter(Boolean).filter((r) => r.verdict === "fail");
257:   const lines = fails.map((r, i) => "[" + (i + 1) + "] (" + r.severity + ") " + r.issues.join(" | "));
258:   return agent(
259:     specBlock() +

262:       "FAILURES:\n" + (lines.join("\n") || "(reviewers failed without detail — re-verify from scratch)") + "\n\n" +
263:       "PREVIOUS ATTEMPT:\n" + (work ? work.summary : "(none)") + "\n\n" +
264:       "Do not stop at a plan. Edit the files and prove the fix with real command output. " +
265:       "If an issue is genuinely unfixable without a user decision, put it in blockedOn.",
266:     { schema: WORK, label: "b" + b.id + ":repair", phase: "repair" }
267:   );

270: // ══ THE BATCH — one pipeline item, no barrier between batches ═══════════════
271: async function runBatch(b) {
272:   log("batch " + b.id + ": gather");
273: 
274:   // 1) Gather x3, parallel.
275:   const gatheredRaw = await parallel([0, 1, 2].map((i) => () => gatherStage(b, i)));
276:   // Tag each surviving gatherer with its seat so debate can exclude the right one.
277:   const gathered = [];
278:   for (let i = 0; i < gatheredRaw.length; i++) {
279:     const g = gatheredRaw[i];
280:     if (g) gathered.push(Object.assign({ __seat: i }, g));
281:   }
282:   if (gathered.length === 0) {
283:     return { id: b.id, title: b.title, status: "failed", reason: "all gatherers failed", iteration: 0 };
284:   }
285: 
286:   // Surface questions the batch could not resolve — Manager handles these.
287:   const openQuestions = [];
288:   for (const g of gathered) for (const q of asArray(g.openQuestions)) if (openQuestions.indexOf(q) === -1) openQuestions.push(q);
289: 
290:   log("batch " + b.id + ": draft ‖ debate");
291:   // 2) Draft and debate genuinely run at the same time: start draft (not awaited),
292:   //    then run the 3 debaters, then collect the draft.
293:   const draftPromise = draftStage(b, gathered);
294:   const debated = (await parallel([0, 1, 2].map((i) => () => debateStage(b, gathered, i)))).filter(Boolean);
295:   const draft = await draftPromise;
296: 
297:   // 3) Final implement.
298:   log("batch " + b.id + ": implement");
299:   let work = await finalStage(b, draft, gathered, debated);
300:   if (!work) {
301:     return { id: b.id, title: b.title, status: "failed", reason: "doer produced no result", iteration: 0, openQuestions };
302:   }
303: 
304:   // 4) Review loop: reviewers + e2e must ALL pass.
305:   let iteration = 0;
306:   let reviews = [];
307:   while (iteration < MAX_ITER) {
308:     iteration += 1;
309:     log("batch " + b.id + ": review round " + iteration);
310:     reviews = (await parallel([0, 1, 2, 3].map((seat) => () => reviewStage(b, work, seat)))).filter(Boolean);
311:     const failures = reviews.filter((r) => r.verdict === "fail");
312:     if (failures.length === 0 && reviews.length === 4) {
313:       return {
314:         id: b.id, title: b.title, status: "clean", iteration,
315:         summary: work.summary, filesTouched: work.filesTouched || [],
316:         howToVerify: work.howToVerify, reviews: reviews.map((r) => ({ verdict: r.verdict, severity: r.severity })),
317:         blockedOn: work.blockedOn || [], openQuestions,
318:       };
319:     }
320:     if (iteration >= MAX_ITER) {
321:       return {
322:         id: b.id, title: b.title, status: "unresolved", iteration,
323:         summary: work.summary, filesTouched: work.filesTouched || [],
324:         remaining: failures.map((f) => ({ severity: f.severity, issues: f.issues })),
325:         reviews: reviews.map((r) => ({ verdict: r.verdict, severity: r.severity })),
326:         blockedOn: work.blockedOn || [], openQuestions,
327:       };
328:     }
329:     // Failed -> hand the failures back to the doer and loop.
330:     log("batch " + b.id + ": repair (round " + iteration + ")");
331:     const repaired = await repairStage(b, work, reviews);
332:     if (!repaired) return { id: b.id, title: b.title, status: "failed", reason: "repair produced no result", iteration, openQuestions };
333:     work = repaired;
334:   }
335:   return { id: b.id, title: b.title, status: "unresolved", reason: "loop exited", iteration, openQuestions };
336: }
